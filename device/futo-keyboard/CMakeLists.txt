cmake_minimum_required(VERSION 3.22)
project(PianoFutoKeyboard VERSION 0.3.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT FUTO_SOURCE OR NOT EXISTS "${FUTO_SOURCE}/NOTICE.piano")
    message(FATAL_ERROR "Prepare verified source with scripts/prepare-futo-keyboard.py first")
endif()
find_package(JNI REQUIRED COMPONENTS JVM) # headers only; no JVM linked at runtime
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Fcitx5Core 5.1.23 REQUIRED)
find_package(Fcitx5Module REQUIRED COMPONENTS DBus)

# Use the upstream's explicit list, excluding Android JNI entrypoints, neural
# models and bundled third-party code. Never glob tests into a release binary.
set(FUTO_JNI "${FUTO_SOURCE}/native/jni")
file(STRINGS "${FUTO_JNI}/NativeFileList.cmake" core_lines
    REGEX "CMAKE_CURRENT_SOURCE_DIR}/src/(dictionary|suggest|utils)/.*\\.cpp")
foreach(line IN LISTS core_lines)
    string(REGEX MATCH "src/[^ )]+\\.cpp" relative "${line}")
    if(NOT relative MATCHES "_test\\.cpp$")
        list(APPEND core_sources "${FUTO_JNI}/${relative}")
    endif()
endforeach()
add_library(futo-core STATIC ${core_sources} futo-core.cpp)
target_include_directories(futo-core PUBLIC "${FUTO_JNI}" "${FUTO_JNI}/src" ${JNI_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(futo-core PRIVATE HAVE_PTHREAD HOST_TOOL)
target_compile_options(futo-core PRIVATE -O2 -fno-strict-aliasing)
target_link_libraries(futo-core PUBLIC Qt6::Core)

add_executable(piano-futo-tool futo-tool.cpp)
target_link_libraries(piano-futo-tool PRIVATE futo-core)
add_library(piano-futo MODULE fcitx-engine.cpp)
target_link_libraries(piano-futo PRIVATE futo-core Fcitx5::Core Fcitx5::Module::DBus)
target_compile_features(piano-futo PRIVATE cxx_std_20)
target_compile_definitions(piano-futo PRIVATE QT_NO_KEYWORDS)
target_compile_options(piano-futo PRIVATE -Wall -Wextra -Werror)
set_target_properties(piano-futo PROPERTIES PREFIX "")
install(TARGETS piano-futo DESTINATION "${CMAKE_INSTALL_LIBDIR}/fcitx5")
install(TARGETS piano-futo-tool RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES piano-futo-addon.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/addon" RENAME piano-futo.conf)
install(FILES piano-futo-im.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod" RENAME piano-futo.conf)
install(FILES "${FUTO_SOURCE}/LICENSE.md" "${FUTO_SOURCE}/NOTICE" "${FUTO_SOURCE}/NOTICE.piano"
    DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/piano-futo-keyboard")
if(BUILD_TESTING AND FUTO_DICTIONARY)
    add_test(NAME futo-native-dictionary COMMAND piano-futo-tool --self-test "${FUTO_DICTIONARY}")
endif()
if(BUILD_TESTING)
    add_executable(piano-futo-engine-test engine-test.cpp)
    target_link_libraries(piano-futo-engine-test PRIVATE futo-core Fcitx5::Core Fcitx5::Module::DBus)
    target_compile_features(piano-futo-engine-test PRIVATE cxx_std_20)
    target_compile_definitions(piano-futo-engine-test PRIVATE QT_NO_KEYWORDS)
    if(FUTO_DICTIONARY)
        add_test(NAME futo-fcitx-integration COMMAND piano-futo-engine-test "${FUTO_DICTIONARY}")
    endif()
endif()
