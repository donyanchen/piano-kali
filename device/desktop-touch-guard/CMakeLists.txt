cmake_minimum_required(VERSION 3.22)
project(ShengDesktopTouchGuard VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus Quick)
find_package(KWin 6.7.4 EXACT REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/touchguard.cpp" guard_source_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/metadata.json" guard_metadata_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" guard_recipe_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/DockRevealHint.qml" guard_hint_hash)
string(SHA256 guard_hash "${guard_source_hash}${guard_metadata_hash}${guard_recipe_hash}${guard_hint_hash}")
string(SUBSTRING "${guard_hash}" 0 12 guard_revision)
set(SHENG_TOUCH_GUARD_ID "sheng_desktop_touch_guard_${guard_revision}")
configure_file(metadata.json "${CMAKE_CURRENT_BINARY_DIR}/metadata.json" @ONLY)
configure_file(touchguard.cpp "${CMAKE_CURRENT_BINARY_DIR}/touchguard.cpp" COPYONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/touch-guard-id.txt" "${SHENG_TOUCH_GUARD_ID}\n")
add_library(${SHENG_TOUCH_GUARD_ID} MODULE "${CMAKE_CURRENT_BINARY_DIR}/touchguard.cpp")
set_target_properties(${SHENG_TOUCH_GUARD_ID} PROPERTIES PREFIX "")
target_link_libraries(${SHENG_TOUCH_GUARD_ID} PRIVATE KWin::kwin Qt6::DBus Qt6::Quick)
qt_add_resources(${SHENG_TOUCH_GUARD_ID} "dock_hint_${guard_revision}"
    PREFIX "/sheng/touchguard/${guard_revision}" FILES DockRevealHint.qml)
target_compile_definitions(${SHENG_TOUCH_GUARD_ID} PRIVATE
    SHENG_GUARD_HINT_URL="qrc:/sheng/touchguard/${guard_revision}/DockRevealHint.qml")
install(TARGETS ${SHENG_TOUCH_GUARD_ID} DESTINATION ${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins)
