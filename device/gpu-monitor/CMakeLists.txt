cmake_minimum_required(VERSION 3.20)
project(sheng_gpu_monitor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)

target_link_libraries(ksystemstats_plugin_gpu PRIVATE Qt6::Core KF6::CoreAddons KSysGuard::SystemStats)
